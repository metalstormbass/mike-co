{{- else if contains "LoadBalancer" .Values.nginx.service.type }}
Get the application URL by running:
  export SERVICE_IP=$(kubectl get svc {{ include "rag-knowledge-base.fullname" . }}-nginx -n {{ .Values.namespace }} -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
  echo "Application URL: http://$SERVICE_IP"
{{- else if contains "NodePort" .Values.nginx.service.type }}
Get the application URL by running:
  export NODE_PORT=$(kubectl get svc {{ include "rag-knowledge-base.fullname" . }}-nginx -n {{ .Values.namespace }} -o jsonpath='{.spec.ports[0].nodePort}')
  export NODE_IP=$(kubectl get nodes -o jsonpath='{.items[0].status.addresses[?(@.type=="ExternalIP")].address}')
  echo "Application URL: http://$NODE_IP:$NODE_PORT"
{{- else }}
Port forward to access the application:
  kubectl port-forward svc/{{ include "rag-knowledge-base.fullname" . }}-nginx 8080:{{ .Values.nginx.service.port }} -n {{ .Values.namespace }}
  echo "Application URL: http://localhost:8080"
{{- end }}
